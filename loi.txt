$files = @(
    "C:\TunerTools\Web\API\api_gp_search.php",
    "C:\TunerTools\Web\API\api_songs.php"
)

foreach ($file in $files) {
    Write-Host "`n=== $(Split-Path $file -Leaf) ===" -ForegroundColor Cyan

    Select-String -LiteralPath $file `
        -SimpleMatch `
        -Pattern @(
            '$tempFilesDir',
            'tuner_images_'
        ) |
    ForEach-Object {

        $line = $_.Line.Trim()

        if ($line -match '(?i)password|token|secret|api.?key|authorization') {
            $line = "[DA AN NOI DUNG NHAY CAM]"
        }

        [PSCustomObject]@{
            Line = $_.LineNumber
            Code = $line
        }
    } |
    Format-Table -AutoSize -Wrap
}
