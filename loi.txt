$server = Invoke-RestMethod `
    "http://127.0.0.1:5000/api/get_cached_songs"

$drive = Invoke-RestMethod `
    "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=cached_songs"

function SongKey($s) {
    $parts = @(
        ([string]$s.source).Trim()
        ([string]$s.id).Trim()
        ([string]$s.title).Trim()
        ([string]$s.singer).Trim()
    )
    return ($parts -join "|").Normalize(
        [Text.NormalizationForm]::FormC
    ).ToLowerInvariant()
}

$serverSongs = @($server.songs)
$driveSongs = @($drive.songs)

$serverKeys = @{}
$driveKeys = @{}

foreach ($s in $serverSongs) {
    $serverKeys[(SongKey $s)] = $true
}

foreach ($s in $driveSongs) {
    $driveKeys[(SongKey $s)] = $true
}

$onlyServer = @($serverSongs | Where-Object {
    -not $driveKeys.ContainsKey((SongKey $_))
})

$onlyDrive = @($driveSongs | Where-Object {
    -not $serverKeys.ContainsKey((SongKey $_))
})

Write-Host "`n===== DOI CHIEU BAI HAT =====" -ForegroundColor Cyan
Write-Host "SERVER TOTAL: $($serverSongs.Count)"
Write-Host "DRIVE TOTAL : $($driveSongs.Count)"
Write-Host "CHI CO SERVER: $($onlyServer.Count)"
Write-Host "CHI CO DRIVE : $($onlyDrive.Count)"

Write-Host "`n===== CHI CO SERVER =====" -ForegroundColor Yellow
$onlyServer |
    Select-Object source,id,title,singer,has_audio_cache,has_lyrics_cache |
    Format-Table -AutoSize

Write-Host "`n===== CHI CO DRIVE (30 BAI DAU) =====" -ForegroundColor Yellow
$onlyDrive |
    Select-Object -First 30 source,id,title,singer,has_audio_cache,has_lyrics_cache |
    Format-Table -AutoSize
