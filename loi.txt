Write-Host "=== KIEM TRA AI ==="

try {
    $result = Invoke-RestMethod `
        "http://127.0.0.1:5000/" `
        -TimeoutSec 15

    Write-Host "AI SERVER ONLINE" -ForegroundColor Green
    Write-Host $result
}
catch {
    Write-Host "AI SERVER ERROR: $($_.Exception.Message)" -ForegroundColor Red
}

Write-Host "`n=== KIEM TRA CONG 5000 ==="

Get-NetTCPConnection -LocalPort 5000 -State Listen `
    -ErrorAction SilentlyContinue |
Select-Object LocalAddress, LocalPort, State, OwningProcess
