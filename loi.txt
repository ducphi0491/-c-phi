$projects = @(
    "C:\audio-sheet-studio",
    "C:\AudioSheetStudio\audio-sheet-studio"
)

foreach ($project in $projects) {
    if (Test-Path "$project\CHAY_WEBSITE.bat") {
        $folder = "$project\web\piano_samples"
        $files = @(Get-ChildItem $folder -Filter "*.mp3" -File -ErrorAction SilentlyContinue)

        Write-Host "`nDu an: $project" -ForegroundColor Cyan
        Write-Host "So mau piano: $($files.Count)/30"

        if ($files.Count -eq 30) {
            Write-Host "[OK] Da co du 30 mau piano!" -ForegroundColor Green
        } else {
            Write-Host "[CHUA XONG] Chua du mau piano." -ForegroundColor Yellow
        }
    }
}
