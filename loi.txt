$names = @(
    "api_songs.php",
    "api_gp_search.php",
    "api_documents.php"
)

$results = foreach ($name in $names) {

    $old = "C:\inetpub\wwwroot\$name"
    $new = "C:\TunerTools\Web\API\$name"

    $oldExists = Test-Path -LiteralPath $old -PathType Leaf
    $newExists = Test-Path -LiteralPath $new -PathType Leaf

    $isWrapper = $false

    if ($oldExists) {

        $content = [IO.File]::ReadAllText($old)

        $expected = "C:/TunerTools/Web/API/$name"

        $isWrapper = $content.Contains($expected)
    }

    [PSCustomObject]@{
        API            = $name
        PHP_Old_Exists = $oldExists
        Wrapper_OK     = $isWrapper
        PHP_New_Exists = $newExists
    }
}

Write-Host "`n=== KET QUA KIEM TRA PHP ===" -ForegroundColor Cyan

$results | Format-Table -AutoSize
