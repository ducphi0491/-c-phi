    foreach ($n in $check.Lines) {
        Write-Host "`n--- DONG $n ---"

        $start = [Math]::Max(1, $n - 1)
        $end = [Math]::Min($content.Count, $n + 1)

        for ($i = $start; $i -le $end; $i++) {
            $line = $content[$i - 1]

            if ($line -match '(?i)password|secret|token|api.?key|authorization|private.?key') {
                $line = "[DA AN NOI DUNG NHAY CAM]"
            }

            Write-Host "${i}: $line"
        }
    }

$root = "C:\TunerTools\Data\GuitarPro"

if (-not (Test-Path -LiteralPath $root -PathType Container)) {
    throw "Khong tim thay thu muc Guitar Pro chinh"
}

[long]$total = 0
[long]$bytes = 0
[long]$gpCount = 0

Get-ChildItem -LiteralPath $root -Recurse -File -ErrorAction Stop |
ForEach-Object {
    $total++
    $bytes += $_.Length

    if ($_.Extension.ToLowerInvariant() -in @(
        ".gp", ".gp3", ".gp4", ".gp5", ".gpx"
    )) {
        $gpCount++
    }
}

Write-Host "`n=== GUITAR PRO HIEN TAI ===" -ForegroundColor Cyan
Write-Host "TONG SO FILE: $total"
Write-Host "FILE GUITAR PRO: $gpCount"
Write-Host "DUNG LUONG BYTE: $bytes"
Write-Host "DUNG LUONG GiB: $([math]::Round($bytes / 1GB, 2))"

if ($total -eq 1639039 -and $bytes -eq 57955266829) {
