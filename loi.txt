
$project = "C:\AudioSheetStudio\audio-sheet-studio"

$zip = Get-ChildItem "$env:USERPROFILE\Downloads\AudioSheet-AI-Diagnostics*.zip" -File -ErrorAction SilentlyContinue |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if ($null -eq $zip) {
    Write-Host "KHONG TIM THAY ZIP. Hay tai ban va vao Downloads." -ForegroundColor Red
} else {
    Expand-Archive -LiteralPath $zip.FullName -DestinationPath $project -Force

    $script = Join-Path $project "scripts\diagnose_ai.py"

    Write-Host "File diagnose_ai.py:" (Test-Path $script)

    if (Test-Path $script) {
        & "$project\.venv\Scripts\python.exe" $script
    }
}
