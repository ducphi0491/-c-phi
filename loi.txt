$zip = Get-ChildItem $downloads `
    -Filter "ScoreOMR_WEB_POWERSHELL_WIN_20261114*.zip" `
    -File |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $zip) {
    throw "Khong tim thay file ZIP trong Downloads."
}

$dir = "C:\TunerTools\Tools\ScoreOMR_Web_Test"

New-Item -ItemType Directory -Path $dir -Force | Out-Null

Expand-Archive -LiteralPath $zip.FullName `
    -DestinationPath $dir -Force -ErrorAction Stop

powershell.exe -NoProfile -ExecutionPolicy Bypass `
    -File "$dir\CHAY_SCORE_OMR_WEB.ps1"
