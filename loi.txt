
$base = "https://api.ez3d.fun/ezmusic"

$models = @(
    "echo-android-1",
    "orpheus-android-1"
)

foreach ($id in $models) {
    try {
        $m = Invoke-RestMethod `
            -Uri "$base/v1/models/$id/current" `
            -Method Get `
            -TimeoutSec 20

        foreach ($f in $m.files) {
            [PSCustomObject]@{
                Model = $id
                File = $f.path
                SizeMB = [math]::Round($f.bytes / 1MB, 2)
                URL = "$base/v1/models/$id/$($m.version)/$($f.path)"
            }
        }
    }
    catch {
        Write-Warning "$id : $($_.Exception.Message)"
    }
}
