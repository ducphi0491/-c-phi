foreach ($profile in @("fluidr3", "musescore")) {
    try {
        $url = "https://tunertools.top/api_gp_search.php?action=soundfont_info&profile=$profile"

        $result = Invoke-RestMethod -Uri $url -TimeoutSec 30

        [PSCustomObject]@{
            Profile = $profile
            Success = $result.success
            File    = $result.data.file_name
        }
    }
    catch {
        Write-Host "ERROR $profile : $($_.Exception.Message)"
    }
}
