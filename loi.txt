$urls = @(
    "http://127.0.0.1:5000/api/get_direct_link",
    "https://api.tunertools.top/api/get_direct_link"
)

foreach ($url in $urls) {
    $output = Join-Path $env:TEMP (
        "tt_direct_link_" + [guid]::NewGuid().ToString("N") + ".txt"
    )

    Write-Host "`n=== $url ===" -ForegroundColor Cyan

    try {
        $status = & curl.exe -sS `
            --max-time 15 `
            --request POST `
            --header "Content-Type: application/json" `
            --data-binary "{}" `
            --output $output `
            --write-out "%{http_code}" `
            $url

        Write-Host "HTTP: $status"
        Write-Host "CURL EXIT: $LASTEXITCODE"

        if (Test-Path -LiteralPath $output) {
            $response = [IO.File]::ReadAllText($output)

            if ($response.Length -gt 300) {
                $response = $response.Substring(0, 300)
            }

            Write-Host "RESPONSE: $response"
        }
    }
    finally {
        Remove-Item -LiteralPath $output -Force -ErrorAction SilentlyContinue
    }
}
