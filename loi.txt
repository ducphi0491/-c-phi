$ErrorActionPreference = "Stop"

$root = "C:\inetpub\wwwroot"
$apiDir = "C:\TunerTools\Web\API"
$backupDir = "C:\TunerTools\Backups"
$phpExe = "C:\php\php.exe"

New-Item -ItemType Directory -Path $apiDir -Force | Out-Null
New-Item -ItemType Directory -Path $backupDir -Force | Out-Null

$enc = [System.Text.Encoding]::GetEncoding(28591)
$stamp = Get-Date -Format "yyyyMMdd_HHmmss"

# Chuan bi hai PHP trong thu muc moi
foreach ($name in @("api_gp_search.php", "api_songs.php")) {

    $source = Join-Path $root $name
    $target = Join-Path $apiDir $name

    if (-not (Test-Path -LiteralPath $source)) {
        throw "Thieu file nguon: $name"
    }

    # Sao luu ban trong thu muc moi neu da ton tai
    if (Test-Path -LiteralPath $target) {
        Copy-Item -LiteralPath $target `
            -Destination (Join-Path $backupDir "$name.$stamp.bak") `
            -ErrorAction Stop
    }

    # Lay ban PHP moi nhat dang hoat dong
    Copy-Item -LiteralPath $source -Destination $target -Force

    $content = $enc.GetString(
        [System.IO.File]::ReadAllBytes($target)
    )

    # Giu nguyen duong dan tai nguyen phu thuoc tam thoi
    $content = $content.Replace(
        '__DIR__',
        "'C:/inetpub/wwwroot'"
    )

    [System.IO.File]::WriteAllBytes(
        $target,
        $enc.GetBytes($content)
    )

    & $phpExe -l $target

    if ($LASTEXITCODE -ne 0) {
        throw "PHP syntax error: $name"
    }

    Write-Host "PREPARED: $name" -ForegroundColor Green
}

# Tao API kiem tra tam thoi
$probeName = "__tt_api_test_$([guid]::NewGuid().ToString('N')).php"
$probePath = Join-Path $root $probeName

$probeCode = "<?php require 'C:/TunerTools/Web/API/api_gp_search.php';"

try {
    [System.IO.File]::WriteAllBytes(
        $probePath,
        [System.Text.Encoding]::ASCII.GetBytes($probeCode)
    )

    $testUrl = "http://127.0.0.1/${probeName}?action=soundfont_info&profile=musescore"

    $response = & curl.exe -sS --max-time 30 `
        -H "Host: tunertools.top" `
        "$testUrl"

    if ($LASTEXITCODE -ne 0) {
        throw "Khong goi duoc API test"
    }

    $result = ($response -join "`n") | ConvertFrom-Json

    if ($result.success -ne $true) {
        throw "API moi chua hoat dong dung"
    }

    Write-Host ""
    Write-Host "NEW API FOLDER TEST OK" -ForegroundColor Green
}
finally {
    if (Test-Path -LiteralPath $probePath) {
        Remove-Item -LiteralPath $probePath -Force
    }
}
