$root = 'C:\inetpub\wwwroot'

$files = @(
    Get-ChildItem $root -File |
    Where-Object { $_.Extension -in '.php', '.config', '.bat' }
)

$files += @(
    Get-ChildItem "$root\tunertools" -Recurse -File |
    Where-Object { $_.Extension -in '.php', '.js', '.json', '.config' }
)

$patterns = @(
    'C:\inetpub\wwwroot',
    'C:\\inetpub\\wwwroot',
    'tab/',
    'MuseScore_General.sf3',
    'FluidR3_GM.sf2'
)

$matches = @($files | Select-String -SimpleMatch -Pattern $patterns)

$matches |
Select-Object Path, LineNumber, Pattern |
Format-Table -AutoSize

Write-Host "TONG SO VI TRI: $($matches.Count)"
