$root = "C:\TunerTools\Data\GuitarPro"
$testDir = Join-Path $root "+"

$file = Get-ChildItem -LiteralPath $testDir -Recurse -File |
    Where-Object {
        $_.Extension -in ".gp3", ".gp4", ".gp5", ".gpx"
    } |
    Select-Object -First 1

if (-not $file) {
    throw "Khong tim thay file GP"
}

$relative = $file.FullName.Substring($root.Length).TrimStart('\')

$encoded = (($relative -split '\\') |
    ForEach-Object { [uri]::EscapeDataString($_) }) -join '/'

$url = "https://tunertools.top/tab/$encoded"

Write-Host "FILE EXISTS: $(Test-Path -LiteralPath $file.FullName)"
Write-Host "TEST URL: $url"

curl.exe --path-as-is -sS -L -r 0-0 -o NUL `
    -w "GP HTTP STATUS: %{http_code}`n" `
    "$url"
