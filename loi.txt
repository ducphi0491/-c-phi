$ErrorActionPreference = "Stop"

$source = "C:\inetpub\wwwroot\img_sys"
$target = "C:\TunerTools\Web\img_sys"
$appcmd = "$env:windir\System32\inetsrv\appcmd.exe"

# 1. Kiem tra du lieu nguon va dich
$files = @(Get-ChildItem -LiteralPath $source -Recurse -File)
$targetFiles = @(Get-ChildItem -LiteralPath $target -Recurse -File)

if ($files.Count -ne $targetFiles.Count) {
    throw "So file khong trung khop"
}

foreach ($f in $files) {
    $relative = $f.FullName.Substring($source.Length).TrimStart('\')
    $dest = Join-Path $target $relative

    if (!(Test-Path -LiteralPath $dest -PathType Leaf)) {
        throw "Thieu file: $relative"
    }

    if ((Get-FileHash -LiteralPath $f.FullName).Hash -ne
        (Get-FileHash -LiteralPath $dest).Hash) {
        throw "File khac nhau: $relative"
    }
}

Write-Host "IMG_SYS HASH VERIFIED: $($files.Count) FILES" -ForegroundColor Green

# 2. Kiem tra URL hien tai
$url = "http://127.0.0.1/img_sys/logo.png"

function Test-ImgUrl {
    $status = (& curl.exe -sS --max-time 20 -r 0-0 `
        -H "Host: tunertools.top" `
        -o NUL -w "%{http_code}" $url | Out-String).Trim()

    if ($LASTEXITCODE -ne 0) {
        throw "Khong ket noi duoc IMG_SYS"
    }

    return $status
}

$before = Test-ImgUrl
Write-Host "HTTP BEFORE: $before"

if ($before -notin @("200", "206")) {
    throw "URL goc chua hoat dong"
}

# 3. Kiem tra anh xa IIS
$vdirs = & $appcmd list vdirs | Out-String

if ($LASTEXITCODE -ne 0) {
    throw "Khong doc duoc IIS"
}

if ($vdirs -match 'VDIR "Default Web Site/img_sys"') {
    throw "IMG_SYS da co Virtual Directory. Khong ghi de."
}

# 4. Backup IIS
$backup = "TT_IMG_SYS_" + (Get-Date -Format "yyyyMMdd_HHmmss")

& $appcmd add backup $backup

if ($LASTEXITCODE -ne 0) {
    throw "Khong backup duoc IIS"
}

$created = $false

try {
    & $appcmd add vdir `
        '/app.name:Default Web Site/' `
        '/path:/img_sys' `
        "/physicalPath:$target"

    if ($LASTEXITCODE -ne 0) {
        throw "Tao Virtual Directory that bai"
    }

    $created = $true

    $after = Test-ImgUrl
    Write-Host "HTTP AFTER: $after"

    if ($after -notin @("200", "206")) {
        throw "URL moi khong hoat dong"
    }

    Write-Host ""
    Write-Host "IMG_SYS MIGRATION OK" -ForegroundColor Green

    & $appcmd list vdirs
}
catch {
    if ($created) {
        & $appcmd delete vdir "Default Web Site/img_sys"
    }

    Write-Host "DA THU KHOI PHUC IIS" -ForegroundColor Yellow
    throw
}
