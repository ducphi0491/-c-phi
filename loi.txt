$ErrorActionPreference = "Stop"

$live = "C:\inetpub\wwwroot\api_gp_search.php"
$newApi = "C:\TunerTools\Web\API\api_gp_search.php"
$backupDir = "C:\TunerTools\Backups"

New-Item -ItemType Directory -Path $backupDir -Force | Out-Null

if (-not (Test-Path -LiteralPath $live -PathType Leaf)) {
    throw "Thieu PHP dang hoat dong."
}

if (-not (Test-Path -LiteralPath $newApi -PathType Leaf)) {
    throw "Thieu PHP tai thu muc moi."
}

function Test-GpApi {
    foreach ($profile in @("fluidr3", "musescore")) {
        $url = "http://127.0.0.1/api_gp_search.php?action=soundfont_info&profile=$profile"

        $response = & curl.exe -fsS --max-time 30 `
            -H "Host: tunertools.top" "$url"

        if ($LASTEXITCODE -ne 0) {
            throw "HTTP API failed: $profile"
        }

        $json = ($response -join "`n") | ConvertFrom-Json

        if ($json.success -ne $true -or
            -not $json.data.file_name) {
            throw "API response invalid: $profile"
        }

        Write-Host "API OK: $profile" -ForegroundColor Green
    }
}

# Kiem tra PHP moi
& "C:\php\php.exe" -l $newApi

if ($LASTEXITCODE -ne 0) {
    throw "PHP moi co loi cu phap."
}

# Kiem tra truoc khi chuyen
Write-Host "`n=== BEFORE ===" -ForegroundColor Cyan
Test-GpApi

$stamp = Get-Date -Format "yyyyMMdd_HHmmss"

$backup = Join-Path $backupDir "api_gp_search_live_$stamp.php.bak"
$stage = Join-Path $backupDir "api_gp_search_wrapper_$stamp.tmp"
$rollbackOld = Join-Path $backupDir "api_gp_search_failed_wrapper_$stamp.bak"

$wrapper = "<?php`r`nrequire 'C:/TunerTools/Web/API/api_gp_search.php';`r`n"

[System.IO.File]::WriteAllText(
    $stage,
    $wrapper,
    [System.Text.Encoding]::ASCII
)

$switched = $false

try {
    # Thay PHP bang wrapper va luu ban goc
    [System.IO.File]::Replace($stage, $live, $backup)
    $switched = $true

    & "C:\php\php.exe" -l $live

    if ($LASTEXITCODE -ne 0) {
        throw "PHP wrapper syntax error."
    }

    Write-Host "`n=== AFTER ===" -ForegroundColor Cyan
    Test-GpApi

    Write-Host ""
    Write-Host "API_GP_SEARCH CHUYEN THANH CONG" -ForegroundColor Green
    Write-Host "Main PHP: $newApi"
    Write-Host "Backup: $backup"
}
catch {
    Write-Host "LOI: $($_.Exception.Message)" -ForegroundColor Red

    if ($switched -and (Test-Path -LiteralPath $backup)) {
        Copy-Item -LiteralPath $backup -Destination $stage -Force

        [System.IO.File]::Replace(
            $stage,
            $live,
            $rollbackOld
        )

        Write-Host "DA KHOI PHUC PHP GOC" -ForegroundColor Yellow
    }

    throw
}
