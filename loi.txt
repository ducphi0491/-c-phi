$ErrorActionPreference = "Stop"

$oldApi = "C:\inetpub\wwwroot\api_documents.php"
$newApi = "C:\TunerTools\Web\API\api_documents.php"

$src = "C:\inetpub\wwwroot\document"
$dst = "C:\TunerTools\Data\Documents"

$marker = "C:\TunerTools\State\Documents\.tunertools_document_schema_v10.ready"

$backupDir = "C:\TunerTools\Backups"
$php = "C:\php\php.exe"
$appcmd = "$env:windir\System32\inetsrv\appcmd.exe"

$url = "https://tunertools.top/api_documents.php?action=course_manifest"

$stamp = Get-Date -Format "yyyyMMdd_HHmmss"
$enc = [Text.Encoding]::GetEncoding(28591)

$createdVdir = $false
$apiChanged = $false

# ============================================
# 1. KIEM TRA TRUOC KHI CHUYEN
# ============================================

foreach ($file in @($oldApi, $newApi, $marker, $php, $appcmd)) {
    if (-not (Test-Path -LiteralPath $file -PathType Leaf)) {
        throw "KHONG TIM THAY FILE: $file"
    }
}

foreach ($dir in @($src, $dst, $backupDir)) {
    if (-not (Test-Path -LiteralPath $dir -PathType Container)) {
        throw "KHONG TIM THAY FOLDER: $dir"
    }
}

& $php -l $newApi

if ($LASTEXITCODE -ne 0) {
    throw "PHP MOI CO LOI CU PHAP."
}

# Kiem tra API hien tai truoc cutover
$baseline = Invoke-WebRequest `
    -Uri $url -UseBasicParsing -TimeoutSec 30

if ($baseline.StatusCode -ne 200) {
    throw "API GOC KHONG TRA VE HTTP 200."
}

# Kiem tra IIS virtual directory
$vdirs = @(& $appcmd list vdir)

if ($LASTEXITCODE -ne 0) {
    throw "KHONG DOC DUOC CAU HINH IIS."
}

$existing = @(
    $vdirs | Where-Object {
        $_ -match '^VDIR "Default Web Site/document" '
    }
)

if ($existing.Count -gt 0) {
    if ($existing[0] -notmatch [regex]::Escape($dst)) {
        throw "IIS /document DA CO CAU HINH KHAC. DUNG CHUYEN."
    }
}

# Backup API goc
$apiBackup = Join-Path $backupDir "api_documents_live_$stamp.php"

Copy-Item -LiteralPath $oldApi `
    -Destination $apiBackup -ErrorAction Stop

$original = $enc.GetString(
    [IO.File]::ReadAllBytes($oldApi)
)

$needle = "declare(strict_types=1);"

if ($original.IndexOf($needle) -lt 0) {
    throw "KHONG TIM THAY PHP DECLARE. DUNG CHUYEN."
}

# ============================================
# 2. CHUAN BI CHE DO TAM CHAN POST
# ============================================

$gate = @'
if (($_SERVER['REQUEST_METHOD'] ?? 'GET') === 'POST') {
    http_response_code(503);
    header('Content-Type: application/json; charset=utf-8');
    header('Retry-After: 60');
    echo '{"success":false,"code":"DOCUMENT_CUTOVER"}';
    exit;
}
'@

$oldGuarded = $original.Replace(
    $needle,
    $needle + "`r`n" + $gate
)

$guardFile = Join-Path $backupDir "documents_guard_$stamp.php"

[IO.File]::WriteAllBytes(
    $guardFile,
    $enc.GetBytes($oldGuarded)
)

& $php -l $guardFile

if ($LASTEXITCODE -ne 0) {
    throw "LOI PHP MAINTENANCE."
}

# Wrapper tam: doc qua API moi, POST van bi chan
$wrapperGuarded = @'
<?php
declare(strict_types=1);

if (($_SERVER['REQUEST_METHOD'] ?? 'GET') === 'POST') {
    http_response_code(503);
    header('Content-Type: application/json; charset=utf-8');
    header('Retry-After: 60');
    echo '{"success":false,"code":"DOCUMENT_CUTOVER"}';
    exit;
}

require 'C:/TunerTools/Web/API/api_documents.php';
'@

# Wrapper chinh thuc
$wrapper = @'
<?php
declare(strict_types=1);

require 'C:/TunerTools/Web/API/api_documents.php';
'@

$wrapperFile = Join-Path $backupDir "documents_wrapper_$stamp.php"
$wrapperGuardFile = Join-Path $backupDir "documents_wrapper_guard_$stamp.php"

[IO.File]::WriteAllText($wrapperFile, $wrapper)
[IO.File]::WriteAllText($wrapperGuardFile, $wrapperGuarded)

foreach ($file in @($wrapperFile, $wrapperGuardFile)) {
    & $php -l $file

    if ($LASTEXITCODE -ne 0) {
        throw "PHP WRAPPER CO LOI: $file"
    }
}

try {

    # ========================================
    # 3. TAM CHAN YEU CAU GHI
    # ========================================

    Copy-Item -LiteralPath $guardFile `
        -Destination $oldApi -Force -ErrorAction Stop

    $apiChanged = $true

    Write-Host "DOCUMENT POST TEMPORARILY BLOCKED" `
        -ForegroundColor Yellow

    Start-Sleep -Seconds 5

    # Kiem tra request dang xu ly
    $activeRequests = @(
        & $appcmd list requests |
        Where-Object { $_ -match 'api_documents\.php' }
    )

    if ($activeRequests.Count -gt 0) {
        throw "CON REQUEST DOCUMENT DANG XU LY. DUNG CUTOVER."
    }

    # ========================================
    # 4. DONG BO DOCUMENT LAN CUOI
    # ========================================

    $log = Join-Path $backupDir "document_final_sync_$stamp.log"

    & robocopy.exe $src $dst `
        /E /COPY:DAT /DCOPY:DAT /XJ `
        /R:2 /W:2 /MT:4 `
        /NP /NFL /NDL "/LOG:$log"

    $rc = $LASTEXITCODE

    if ($rc -ge 8) {
        throw "ROBOCOPY LOI: $rc"
    }

    # Kiem tra tung duong dan va dung luong file
    $srcList = @(
        Get-ChildItem -LiteralPath $src -Recurse -File -Force |
        ForEach-Object {
            $_.FullName.Substring($src.Length).ToLowerInvariant() +
            "|" + $_.Length
        } | Sort-Object
    )

    $dstList = @(
        Get-ChildItem -LiteralPath $dst -Recurse -File -Force |
        ForEach-Object {
            $_.FullName.Substring($dst.Length).ToLowerInvariant() +
            "|" + $_.Length
        } | Sort-Object
    )

    $diff = @(
        Compare-Object -ReferenceObject $srcList `
            -DifferenceObject $dstList
    )

    if ($diff.Count -ne 0) {
        throw "DOCUMENT SOURCE/DEST CO CHENH LECH."
    }

    Write-Host "DOCUMENT FINAL SYNC OK: $($srcList.Count) FILES" `
        -ForegroundColor Green

    # ========================================
    # 5. CHUYEN IIS /document
    # ========================================

    if ($existing.Count -eq 0) {

        & $appcmd add vdir `
            "/app.name:Default Web Site/" `
            "/path:/document" `
            "/physicalPath:$dst"

        if ($LASTEXITCODE -ne 0) {
            throw "KHONG TAO DUOC IIS /document."
        }

        $createdVdir = $true
    }

    Write-Host "IIS /document READY" -ForegroundColor Green

    # ========================================
    # 6. CHUYEN API - VAN CHAN POST
    # ========================================

    Copy-Item -LiteralPath $wrapperGuardFile `
        -Destination $oldApi -Force -ErrorAction Stop

    $result = Invoke-WebRequest `
        -Uri $url -UseBasicParsing -TimeoutSec 30

    if ($result.StatusCode -ne 200) {
        throw "API MOI KHONG TRA HTTP 200."
    }

    $json = $result.Content | ConvertFrom-Json -ErrorAction Stop

    if ($json.success -eq $false) {
        throw "API MOI TRA VE SUCCESS FALSE."
    }

    Write-Host "NEW DOCUMENT API HTTP 200 / JSON OK" `
        -ForegroundColor Green

    # ========================================
    # 7. MO LAI API CHINH THUC
    # ========================================

    Copy-Item -LiteralPath $wrapperFile `
        -Destination $oldApi -Force -ErrorAction Stop

    Write-Host ""
    Write-Host "===================================" -ForegroundColor Green
    Write-Host "DOCUMENT BACKEND CUTOVER OK" -ForegroundColor Green
    Write-Host "===================================" -ForegroundColor Green

    Write-Host "API: C:\TunerTools\Web\API\api_documents.php"
    Write-Host "DATA: C:\TunerTools\Data\Documents"
    Write-Host "IIS: /document"
    Write-Host "BACKUP: $apiBackup"
    Write-Host "SOURCE DATA: KEPT"
}
catch {

    Write-Host "`nCUTOVER FAILED - STARTING ROLLBACK" `
        -ForegroundColor Red

    if ($apiChanged) {
        Copy-Item -LiteralPath $apiBackup `
            -Destination $oldApi -Force
    }

    if ($createdVdir) {
        & $appcmd delete vdir "Default Web Site/document"

        if ($LASTEXITCODE -ne 0) {
            Write-Warning "CAN KIEM TRA LAI IIS /document."
        }
    }

    Write-Host "OLD API RESTORED" -ForegroundColor Yellow

    throw
}
