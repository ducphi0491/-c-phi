$source = "C:\inetpub\wwwroot\tunertools"
$target = "C:\TunerTools\Web\tunertools"
$appcmd = "$env:windir\System32\inetsrv\appcmd.exe"

if (-not (Test-Path -LiteralPath $source -PathType Container)) {
    throw "Khong tim thay website goc."
}

New-Item -ItemType Directory -Path $target -Force | Out-Null

Write-Host "`n=== DONG BO WEBSITE ===" -ForegroundColor Cyan

& robocopy.exe $source $target /E /COPY:DAT /DCOPY:DAT /R:2 /W:1 /NFL /NDL /NP

if ($LASTEXITCODE -ge 8) {
    throw "Robocopy gap loi."
}

Write-Host "`n=== KIEM TRA SO FILE ===" -ForegroundColor Cyan

$sourceCount = (Get-ChildItem $source -Recurse -File | Measure-Object).Count
$targetCount = (Get-ChildItem $target -Recurse -File | Measure-Object).Count

Write-Host "SOURCE FILES: $sourceCount"
Write-Host "TARGET FILES: $targetCount"

Write-Host "`n=== IIS SITES ===" -ForegroundColor Cyan
& $appcmd list sites

Write-Host "`n=== IIS APPLICATIONS ===" -ForegroundColor Cyan
& $appcmd list apps

Write-Host "`n=== IIS VIRTUAL DIRECTORIES ===" -ForegroundColor Cyan
& $appcmd list vdirs

Write-Host "`n=== WEBSITE FOLDERS ===" -ForegroundColor Cyan

Get-ChildItem -LiteralPath $target -Directory |
    Select-Object Name |
    Format-Table -AutoSize

Write-Host "`nKIEM TRA HOAN TAT" -ForegroundColor Green
