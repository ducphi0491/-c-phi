    @{
        File = "C:\TunerTools\Web\API\api_gp_search.php"
        Lines = @(1358, 3474, 3498, 3557)
    },
    @{

foreach ($check in $checks) {
    Write-Host "`n=== $($check.File) ===" -ForegroundColor Cyan

    $content = Get-Content -LiteralPath $check.File

    foreach ($n in $check.Lines) {
        Write-Host "`n--- DONG $n ---"

        $start = [Math]::Max(1, $n - 1)
        $end = [Math]::Min($content.Count, $n + 1)

        for ($i = $start; $i -le $end; $i++) {

            if ($line -match '(?i)password|secret|token|api.?key|authorization|private.?key') {
                $line = "[DA AN NOI DUNG NHAY CAM]"
            }

            Write-Host "${i}: $line"
        }
    }
