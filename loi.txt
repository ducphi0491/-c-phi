    -Filter "TunerTools_WEB_V6_1_FIX_XUAT_DOI_CHIEU*.zip" `
    -File |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $zip) {
    throw "Khong tim thay ZIP V6.1 trong Downloads."
}

$folder = "$downloads\TunerTools_V6_1_FIX"

Expand-Archive -LiteralPath $zip.FullName `
    -DestinationPath $folder -Force -ErrorAction Stop

powershell.exe -NoProfile -ExecutionPolicy Bypass `
    -File "$folder\CAI_DAT_V6.ps1"

if ($LASTEXITCODE -ne 0) {
    throw "Cai dat that bai."
}

& "C:\TunerTools\Tools\PDF_TAB_GP_WEB\CHAY_WEB.bat"
