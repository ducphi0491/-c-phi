$folders = @(
    @{
        Name = "THU MUC MOI"
        Path = "C:\TunerTools\Data\support_chat_uploads"
    },
    @{
        Name = "THU MUC CU"
        Path = "C:\inetpub\wwwroot\support_chat_uploads"
    }
)

foreach ($folder in $folders) {
    $files = @(Get-ChildItem -LiteralPath $folder.Path -Recurse -File)
    $latest = $files |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1

    [PSCustomObject]@{
        Folder = $folder.Name
        Files = $files.Count
        LastUpdated = if ($latest) {
            $latest.LastWriteTime.ToString("yyyy-MM-dd HH:mm:ss")
        } else {
            "-"
        }
    }
}
