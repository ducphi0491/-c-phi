$source = "C:\inetpub\wwwroot"
$target = "C:\TunerTools\Web\API"

New-Item -ItemType Directory -Path $target -Force | Out-Null

$files = @(
    "api_songs.php",
    "api_gp_search.php",
    "api_AI.bat"
)

foreach ($file in $files) {
    $old = Join-Path $source $file
    $new = Join-Path $target $file

    Copy-Item -LiteralPath $old -Destination $new -Force -ErrorAction Stop

    $hashOld = (Get-FileHash $old -Algorithm SHA256).Hash
    $hashNew = (Get-FileHash $new -Algorithm SHA256).Hash

    [PSCustomObject]@{
        File = $file
        Identical = ($hashOld -eq $hashNew)
    }
}
