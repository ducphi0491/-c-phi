$r = Invoke-RestMethod `
  "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=cached_songs" `
  -TimeoutSec 120

[PSCustomObject]@{
    Nguon = $r.source
    TongBai = @($r.songs).Count
    CoAmThanh = @(
        $r.songs | Where-Object {
            $_.has_audio_cache -eq $true
        }
    ).Count
    CoLoi = @(
        $r.songs | Where-Object {
            $_.has_lyrics_cache -eq $true
        }
    ).Count
    CoCaAmThanhVaLoi = @(
        $r.songs | Where-Object {
            $_.has_audio_cache -eq $true -and
            $_.has_lyrics_cache -eq $true
        }
    ).Count
} | Format-List
