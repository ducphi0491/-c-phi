$url = "https://tunertools.top/api_songs.php?action=ai_cached_songs"

try {
    $response = Invoke-WebRequest `
        -Uri $url -UseBasicParsing -TimeoutSec 120

    Write-Host "HTTP STATUS:" $response.StatusCode
    Write-Host "CONTENT TYPE:" $response.Headers["Content-Type"]

    $body = [string]$response.Content
    Write-Host "BODY LENGTH:" $body.Length

    if ($body.Length -gt 0) {
        Write-Host "RESPONSE:"
        Write-Host $body.Substring(
            0, [Math]::Min(1200, $body.Length)
        )
    } else {
        Write-Host "RESPONSE BODY IS EMPTY"
    }
} catch {
    Write-Host "ERROR:" $_.Exception.Message
}
