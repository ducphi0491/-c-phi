$routes = @(
    "api/analyze_file",
    "api/get_lyrics"
)

foreach ($route in $routes) {
    $url = "https://api.tunertools.top/$route"

    Write-Host "`n=== $url ===" -ForegroundColor Cyan

    & curl.exe -sS --max-time 20 `
        -o NUL `
        -w "HTTP STATUS: %{http_code}`n" `
        $url
}
