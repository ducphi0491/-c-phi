$root = "C:\TunerTools\Audio\AI_Music"

foreach ($name in @("ZING_AUDIO", "YOUTUBE_CACHE", "audio_cache", "ZING_JSON")) {
    $path = Join-Path $root $name
    Write-Host "`n===== $name =====" -ForegroundColor Cyan

    if (Test-Path $path) {
        Get-ChildItem $path -File -ErrorAction SilentlyContinue |
        Where-Object {
            $_.Extension -in @(".ogg", ".wav", ".mp3", ".m4a", ".json")
        } |
        Select-Object -First 5 Name, Length, LastWriteTime
    }
}


$file = Get-ChildItem "C:\TunerTools\Audio\AI_Music\ZING_AUDIO" `
    -File -Filter "zing_acapella_*.ogg" |
    Select-Object -First 1

if ($file) {
    $url = "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=lookup&filename=$([uri]::EscapeDataString($file.Name))"
    Invoke-RestMethod -Uri $url | Format-List
} else {
    Write-Host "Chua tim thay file Zing OGG."
}
