$file = "C:\inetpub\wwwroot\api_gp_search.php"
$lines = Get-Content -LiteralPath $file

$positions = @(415, 420, 766, 853)

foreach ($n in $positions) {
    Write-Host "`n========== DONG $n ==========" -ForegroundColor Cyan

    $start = [Math]::Max(1, $n - 3)
    $end = [Math]::Min($lines.Count, $n + 3)

    for ($i = $start; $i -le $end; $i++) {
        $content = $lines[$i - 1]

        if ($content -match '(?i)token|password|secret|api_key|authorization') {
            $content = "[DA AN DONG NHAY CAM]"
        }

        Write-Host "${i}: $content"
    }
}
