$ErrorActionPreference = "Stop"

$src = "C:\inetpub\wwwroot\document"
$dst = "C:\TunerTools\Data\Documents"
$logDir = "C:\TunerTools\Backups"

if (-not (Test-Path -LiteralPath $src -PathType Container)) {
    throw "KHONG TIM THAY THU MUC TAI LIEU GOC."
}

# Khong ghi de du lieu dich da co
if (Test-Path -LiteralPath $dst) {
    $existing = Get-ChildItem -LiteralPath $dst -Force |
        Select-Object -First 1

    if ($existing) {
        throw "THU MUC DICH DA CO DU LIEU. CAN KIEM TRA TRUOC."
    }
}

$sourceSize = (
    Get-ChildItem -LiteralPath $src -Recurse -File -Force |
    Measure-Object -Property Length -Sum
).Sum

$free = (Get-PSDrive C).Free

if ($free -lt ($sourceSize + 8GB)) {
    throw "DUNG LUONG O C KHONG DU DE COPY AN TOAN."
}

New-Item -ItemType Directory -Path $dst -Force | Out-Null
New-Item -ItemType Directory -Path $logDir -Force | Out-Null

$stamp = Get-Date -Format "yyyyMMdd_HHmmss"
$log = Join-Path $logDir "document_copy_$stamp.log"

Write-Host "`nBAT DAU COPY DOCUMENT..." -ForegroundColor Cyan

& robocopy.exe $src $dst `
    /E /COPY:DAT /DCOPY:DAT /XJ `
    /R:2 /W:2 /MT:4 `
    /NP /NFL /NDL /TEE "/LOG:$log"

$rc = $LASTEXITCODE

if ($rc -ge 8) {
    throw "ROBOCOPY GAP LOI. EXIT CODE: $rc. XEM LOG: $log"
}

Write-Host "`n=== DOI CHIEU DU LIEU ===" -ForegroundColor Cyan

foreach ($dir in @($src, $dst)) {
    $stats = Get-ChildItem -LiteralPath $dir `
        -Recurse -File -Force |
        Measure-Object -Property Length -Sum

    [PSCustomObject]@{
        Folder = $dir
        Files = $stats.Count
        Bytes = [long]$stats.Sum
        SizeGB = [math]::Round($stats.Sum / 1GB, 2)
    }
}

Write-Host "`nROBOCOPY EXIT CODE: $rc"
Write-Host "LOG: $log"
Write-Host "KHONG XOA DU LIEU GOC." -ForegroundColor Yellow
