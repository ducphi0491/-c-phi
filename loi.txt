$root = "C:\inetpub\wwwroot"

$names = @("guitar_HD", "img_sys", "libs")

Write-Host "`n=== DUNG LUONG DU LIEU ===" -ForegroundColor Cyan

foreach ($name in $names) {
    $path = Join-Path $root $name

    $stats = Get-ChildItem -LiteralPath $path -Recurse -File |
        Measure-Object -Property Length -Sum

    [PSCustomObject]@{
        Folder = $name
        Files = $stats.Count
        SizeMB = [math]::Round($stats.Sum / 1MB, 2)
    }
}

Write-Host "`n=== KIEM TRA URL ===" -ForegroundColor Cyan

$urls = @(
    "/guitar_HD/course_index.json",
    "/libs/guitar.json",
    "/libs/piano.json",
    "/libs/ukulele.json",
    "/img_sys/logo.png"
)

foreach ($path in $urls) {
    $url = "http://127.0.0.1$path"

    $status = & curl.exe -sS --max-time 15 `
        -H "Host: tunertools.top" `
        -r 0-0 -o NUL `
        -w "%{http_code}" "$url"

    [PSCustomObject]@{
        URL = $path
        HTTP = $status
    }
}
