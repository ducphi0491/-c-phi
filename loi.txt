$ErrorActionPreference = "Stop"

$source = "C:\inetpub\wwwroot\support_chat_uploads"
$target = "C:\TunerTools\Data\support_chat_uploads"

if (-not (Test-Path -LiteralPath $source -PathType Container)) {
    throw "Khong tim thay thu muc upload goc."
}

New-Item -ItemType Directory -Path $target -Force | Out-Null

Write-Host "`n=== SAO CHEP DU LIEU ===" -ForegroundColor Cyan

& robocopy.exe $source $target /E /COPY:DAT /DCOPY:DAT /R:2 /W:1 /NFL /NDL /NP

if ($LASTEXITCODE -ge 8) {
    throw "Robocopy that bai."
}

Write-Host "`n=== KIEM TRA FILE ===" -ForegroundColor Cyan

$files = @(Get-ChildItem -LiteralPath $source -Recurse -File)
$targetFiles = @(Get-ChildItem -LiteralPath $target -Recurse -File)

if ($files.Count -ne $targetFiles.Count) {
    throw "So file khong trung khop."
}

foreach ($f in $files) {
    $relative = $f.FullName.Substring($source.Length).TrimStart('\')
    $dest = Join-Path $target $relative

    if (-not (Test-Path -LiteralPath $dest -PathType Leaf)) {
        throw "Thieu file: $relative"
    }

    if ((Get-FileHash -LiteralPath $f.FullName).Hash -ne
        (Get-FileHash -LiteralPath $dest).Hash) {
        throw "Hash khong khop: $relative"
    }
}

Write-Host "UPLOAD COPY VERIFIED: $($files.Count) FILES" -ForegroundColor Green

Write-Host "`n=== KIEM TRA DUONG DAN PHP ===" -ForegroundColor Cyan

foreach ($name in @("api_gp_search.php", "api_songs.php")) {
    $file = "C:\TunerTools\Web\API\$name"

    Select-String -LiteralPath $file -SimpleMatch -Pattern "support_chat_uploads" |
    Where-Object { $_.Line -match '\$baseDir\s*=' } |
    ForEach-Object {
        Write-Host "$name : $($_.Line.Trim())"
    }
}

Write-Host "`nHOAN THANH BUOC CHUAN BI" -ForegroundColor Green
