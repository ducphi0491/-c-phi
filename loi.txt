$root = "C:\inetpub\wwwroot"

Write-Host "`n=== KIEM TRA URL IMG_SYS ===" -ForegroundColor Cyan

$status = & curl.exe -sS --max-time 15 `
    -H "Host: tunertools.top" `
    -o NUL -w "%{http_code}" `
    "http://127.0.0.1/img_sys/logo.png"

Write-Host "IMG_SYS HTTP: $status"

Write-Host "`n=== TIM HE THONG SU DUNG IMG_SYS ===" -ForegroundColor Cyan

$folders = @(
    "$root\tunertools",
    "$root\document",
    "$root\melowave"
)

$files = @(
    Get-ChildItem -LiteralPath $root -File -ErrorAction SilentlyContinue
)

foreach ($folder in $folders) {
    if (Test-Path -LiteralPath $folder) {
        $files += Get-ChildItem -LiteralPath $folder `
            -Recurse -File -ErrorAction SilentlyContinue
    }
}

$files = $files | Where-Object {
    $_.Extension -in ".php", ".js", ".json", ".html", ".css", ".config" `
    -and $_.Length -lt 2MB
}

$matches = @(
    $files | Select-String -SimpleMatch `
        -Pattern "img_sys/", "img_sys\" `
        -ErrorAction SilentlyContinue
)

if ($matches.Count -eq 0) {
    Write-Host "KHONG TIM THAY THAM CHIEU TRONG CAC FILE DA QUET"
}
else {
    $matches |
        Select-Object Path, LineNumber |
        Format-Table -AutoSize
}

Write-Host "`n=== KIEM TRA BAN SAO ===" -ForegroundColor Cyan

Test-Path "C:\TunerTools\Web\img_sys\logo.png"
