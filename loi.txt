$root = "C:\TunerTools\Audio\AI_Music"

Write-Host "`n=== TIM API GET_LYRICS ===" -ForegroundColor Cyan

$files = @(
    Get-ChildItem -LiteralPath $root -Recurse -File `
        -Filter "*.py" -ErrorAction Stop |
    Where-Object {
        $_.FullName -notmatch '(?i)[\\/](venv|\.venv|site-packages)[\\/]'
    }
)

$matches = @(
    $files | Select-String -SimpleMatch -Pattern "get_lyrics"
)

if ($matches.Count -eq 0) {
    Write-Host "KHONG TIM THAY GET_LYRICS TRONG PYTHON" `
        -ForegroundColor Yellow
}
else {
    $matches |
        Select-Object Path, LineNumber |
        Format-Table -AutoSize
}

$appcmd = "$env:windir\System32\inetsrv\appcmd.exe"

& $appcmd list site
