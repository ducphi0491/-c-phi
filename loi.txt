
$ErrorActionPreference = "Stop"

# Bat TLS 1.2 de tai file
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

# Thu muc du an
$app = "C:\TunerTools\Tools\ScoreOMR_Web_Test"
$jdkDir = Join-Path $app "JDK21"
$zip = Join-Path $app "microsoft-jdk-21.zip"

# Link Microsoft OpenJDK 21 x64 chinh thuc
$url = "https://aka.ms/download-jdk/microsoft-jdk-21.0.13.0.2-windows-x64.zip"

if (-not (Test-Path $app)) {
    throw "Khong tim thay thu muc ScoreOMR_Web_Test."
}

New-Item -ItemType Directory -Path $jdkDir -Force | Out-Null

# Tai Java JDK 21
if (-not (Test-Path $zip)) {
    Write-Host "Dang tai Microsoft JDK 21..." -ForegroundColor Cyan
    Invoke-WebRequest -Uri $url -OutFile $zip -UseBasicParsing
}

# Xac minh SHA-256 voi Microsoft
$checksumUrl = "$url.sha256sum.txt"
$checksumText = (Invoke-WebRequest -Uri $checksumUrl -UseBasicParsing).Content
$expected = [regex]::Match($checksumText, '(?i)\b[a-f0-9]{64}\b').Value

if (-not $expected) {
    throw "Khong doc duoc SHA-256 tu Microsoft."
}

$actual = (Get-FileHash -LiteralPath $zip -Algorithm SHA256).Hash

if ($actual -ne $expected) {
    throw "File JDK sai SHA-256. Hay xoa ZIP va tai lai."
}

# Giai nen Java neu chua co
$javac = Get-ChildItem $jdkDir -Filter javac.exe -Recurse -File |
    Select-Object -First 1

if (-not $javac) {
    Write-Host "Dang giai nen JDK 21..." -ForegroundColor Cyan
    Expand-Archive -LiteralPath $zip -DestinationPath $jdkDir -Force

    $javac = Get-ChildItem $jdkDir -Filter javac.exe -Recurse -File |
        Select-Object -First 1
}

if (-not $javac) {
    throw "Khong tim thay javac.exe sau khi giai nen."
}

# Chon Java 21 chi cho cua so PowerShell nay
$env:JAVA_HOME = Split-Path (Split-Path $javac.FullName -Parent) -Parent
$env:Path = (Join-Path $env:JAVA_HOME "bin") + ";" + $env:Path

Write-Host "JAVA_HOME = $env:JAVA_HOME" -ForegroundColor Green

java -version
javac -version

# Khoi dong Score-OMR
$run = Join-Path $app "CHAY_SCORE_OMR_WEB.ps1"

powershell.exe -NoProfile -ExecutionPolicy Bypass -File $run
