$manifestPath = "C:\TunerTools\BackupPrep\Drive_AppGuides\app_guides_drive_manifest.json"

if (-not (Test-Path -LiteralPath $manifestPath)) {
    throw "Khong tim thay manifest: $manifestPath"
}

$m = Get-Content -LiteralPath $manifestPath -Raw -Encoding UTF8 |
    ConvertFrom-Json

Write-Host "`n=== 14 CANH BAO ===" -ForegroundColor Cyan

$m.warnings | ForEach-Object {
    Write-Host $_ -ForegroundColor Yellow
}

Write-Host "`n=== KIEM TRA TUNG VIDEO ===" -ForegroundColor Cyan

$m.files |
    Group-Object {
        ($_.relative_path.Split('/')[0..1] -join '/')
    } |
    ForEach-Object {
        $items = @($_.Group)

        [PSCustomObject]@{
            BaiHuongDan = $_.Name
            MP4 = @($items | Where-Object { $_.type -eq 'video' }).Count
            VTT = @($items | Where-Object { $_.type -eq 'subtitle' }).Count
            JSON = @($items | Where-Object { $_.type -eq 'metadata' }).Count
            Anh = @($items | Where-Object { $_.type -eq 'thumbnail' }).Count
        }
    } |
    Format-Table -AutoSize
