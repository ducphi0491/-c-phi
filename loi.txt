$folder = "C:\TunerTools\Audio\AI_Music\ZING_AUDIO"

$all = @(Get-ChildItem -LiteralPath $folder -File -ErrorAction Stop)
$valid = @($all | Where-Object {
    $_.Name -match "^zing_acapella_([A-Za-z0-9_-]{6,64})\.(ogg|mp3|wav)$"
})

Write-Host "Tong file ZING_AUDIO: $($all.Count)"
Write-Host "File dung mau ten va co ID: $($valid.Count)"

foreach ($file in ($valid | Select-Object -First 5)) {
    $encoded = [uri]::EscapeDataString($file.Name)
    $url = "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=lookup&filename=$encoded"

    try {
        $result = Invoke-RestMethod -Uri $url -TimeoutSec 30
        [PSCustomObject]@{
            File = $file.Name
            DriveFound = $result.found
            Source = $result.source
            LocalBytes = $file.Length
            DriveBytes = $result.size_bytes
        } | Format-List
    } catch {
        Write-Host "LOI: $($file.Name) - $($_.Exception.Message)"
    }
}
