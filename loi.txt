
$downloads = "$env:USERPROFILE\Downloads"

$zip = Get-ChildItem $downloads `
    -Filter "TunerTools_WEB_V6_TAB_FIX_20261114*.zip" `
    -File |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $zip) {
    throw "Khong tim thay ZIP V6 trong Downloads."
}

$folder = "$downloads\TunerTools_V6"

Expand-Archive -LiteralPath $zip.FullName `
    -DestinationPath $folder -Force

powershell.exe -NoProfile -ExecutionPolicy Bypass `
    -File "$folder\CAI_DAT_V6.ps1"

if ($LASTEXITCODE -ne 0) {
    throw "Cai dat V6 that bai."
}

& "C:\TunerTools\Tools\PDF_TAB_GP_WEB\CHAY_WEB.bat"
