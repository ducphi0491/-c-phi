$files = @(
    "C:\inetpub\wwwroot\api_gp_search.php",
    "C:\inetpub\wwwroot\api_songs.php"
)

foreach ($file in $files) {
    Write-Host "`nFILE: $file" -ForegroundColor Cyan

    Select-String -Path $file -Pattern `
        '__DIR__',
        'DOCUMENT_ROOT',
        'realpath',
        'file_name',
        'DIRECTORY_SEPARATOR',
        'Vietnam',
        'dieu_dem_nhac',
        'SoundFont' |
    Select-Object LineNumber, @{
        Name = 'Code'
        Expression = {
            if ($_.Line -match '(?i)password|token|secret|api.?key|authorization') {
                '[HIDDEN]'
            } else {
                $_.Line.Trim()
            }
        }
    } |
    Format-Table -AutoSize -Wrap
}
