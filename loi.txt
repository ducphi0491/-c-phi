$url = "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=cached_songs"

$d = Invoke-RestMethod -Uri $url -TimeoutSec 120
$songs = @($d.songs)

$audio = @($songs | Where-Object {
    $_.has_audio_cache -eq $true
})

$both = @($songs | Where-Object {
    $_.has_audio_cache -eq $true -and
    $_.has_lyrics_cache -eq $true
})

$audioOnly = @($songs | Where-Object {
    $_.has_audio_cache -eq $true -and
    $_.has_lyrics_cache -ne $true
})

Write-Host "`n===== NGUON GOC SO 1280 =====" -ForegroundColor Cyan

Write-Host "AUDIO FILES IN STREAM_CACHE : $($d.audio_entries_scanned)"
Write-Host "JSON FILES IN LYRICS       : $($d.json_entries_scanned)"
Write-Host "SONGS HAS AUDIO            : $($audio.Count)"
Write-Host "SONGS HAS AUDIO + JSON     : $($both.Count)"
Write-Host "SONGS AUDIO ONLY           : $($audioOnly.Count)"

Write-Host "`n===== CAC BAI CHI CO AUDIO =====" -ForegroundColor Yellow

$audioOnly |
    Select-Object source,id,title,singer,has_audio_cache,has_lyrics_cache |
    Format-Table -AutoSize
