$oldPath = 'C:\AI_Music'

$folders = @(
    'C:\inetpub\wwwroot',
    'C:\TunerTools\Audio\AI_Music',
    'C:\inetpub\wwwroot\tunertools'
)

$extensions = @('.php', '.py', '.bat', '.cmd', '.ps1', '.vbs', '.config')

$files = @()

foreach ($folder in $folders) {
    if (Test-Path $folder) {
        $files += Get-ChildItem $folder -File -ErrorAction SilentlyContinue |
            Where-Object { $_.Extension -in $extensions }
    }
}

foreach ($folder in $folders[1..2]) {
    if (Test-Path $folder) {
        $files += Get-ChildItem $folder -Recurse -File -ErrorAction SilentlyContinue |
            Where-Object { $_.Extension -in $extensions }
    }
}

$matches = @(
    $files |
    Sort-Object FullName -Unique |
    Select-String -SimpleMatch $oldPath -ErrorAction SilentlyContinue
)

if ($matches.Count -eq 0) {
    Write-Host "KHONG TIM THAY DUONG DAN CU TRONG CAC FILE DA KIEM TRA" -ForegroundColor Green
} else {
    $matches |
        Select-Object Path, LineNumber |
        Format-Table -AutoSize
}

Write-Host "SO VI TRI THAM CHIEU DUONG DAN CU: $($matches.Count)"
