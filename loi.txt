$file = "C:\inetpub\wwwroot\api_gp_search.php"

Select-String -Path $file -SimpleMatch -Pattern @(
    '__DIR__',
    'DOCUMENT_ROOT',
    'realpath',
    'dirname(',
    'file_name',
    'soundfont',
    'tabRoot',
    'tabDir'
) |
Where-Object {
    $_.Line -notmatch '(?i)password|token|secret|api_key|authorization'
} |
Select-Object LineNumber,
@{Name="Code";Expression={$_.Line.Trim()}} |
Format-Table -AutoSize -Wrap
