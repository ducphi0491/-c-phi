$urls = @{
    "SERVER" = "http://127.0.0.1:5000/api/get_cached_songs"
    "DRIVE"  = "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=cached_songs"
}

foreach ($name in $urls.Keys) {
    $d = Invoke-RestMethod -Uri $urls[$name] -TimeoutSec 120
    $songs = @($d.songs)

    Write-Host "`n===== $name =====" -ForegroundColor Cyan
    Write-Host "TOTAL SONGS: $($songs.Count)"

    $songs |
        Where-Object { $_.title -like "*Phố Hoa*" } |
        Select-Object id,title,singer |
        Format-Table -AutoSize
}
