>> function SongKey($s) {
>>     $parts = @(
>>         ([string]$s.source).Trim()
>>         ([string]$s.id).Trim()
>>         ([string]$s.title).Trim()
>>         ([string]$s.singer).Trim()
>>     )
>>     return ($parts -join "|").Normalize(
>>         [Text.NormalizationForm]::FormC
