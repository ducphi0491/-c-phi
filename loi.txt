$names = @(
    "gp_fresh_1600000.db",
    "gp_fresh_1600000_backup.db"
)

foreach ($name in $names) {
    $source = "C:\TunerToolsGP_Fresh\$name"
    $target = "C:\TunerTools\Tools\GP_Fresh\$name"

    $hash1 = (Get-FileHash -LiteralPath $source -Algorithm SHA256).Hash
    $hash2 = (Get-FileHash -LiteralPath $target -Algorithm SHA256).Hash

    [PSCustomObject]@{
        File = $name
        Identical = ($hash1 -eq $hash2)
    }
}
