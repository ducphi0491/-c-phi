$filename = "18JJ0AEDF3CO_Dòng Sông Mùa Thu_Nguyễn Hải Yến.json"
$encoded = [uri]::EscapeDataString($filename)

$url = "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=lyrics&filename=$encoded"

$json = Invoke-RestMethod -Uri $url -TimeoutSec 40

[PSCustomObject]@{
    ZingErrorCode = $json.err
    SoCauKaraoke = @($json.data.sentences).Count
    CoDuLieuLoi = (@($json.data.sentences).Count -gt 0)
} | Format-List


$r = Invoke-RestMethod `
  "https://tuner.alwaysdata.net/api_ai_media_drive.php?action=cached_songs" `
  -TimeoutSec 120

$r | Select-Object ok, source, count,
    audio_entries_scanned, json_entries_scanned

$r.songs |
    Where-Object { $_.id -eq "18JJ0AEDF3CO" } |
    Select-Object id, title, singer,
        has_audio_cache, has_lyrics_cache |
    Format-Table -AutoSize
