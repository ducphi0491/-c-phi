$ErrorActionPreference = "Stop"

$file = "C:\inetpub\wwwroot\api_songs.php"
$dest = "C:\TunerTools\Web\API\api_songs.php"
$gpFolder = "C:\TunerTools\Data\GuitarPro"

if (-not (Test-Path -LiteralPath $gpFolder -PathType Container)) {
    throw "Khong tim thay thu muc Guitar Pro moi."
}

$encoding = [System.Text.Encoding]::GetEncoding(28591)
$content = $encoding.GetString(
    [System.IO.File]::ReadAllBytes($file)
)

$old = '$baseDir = __DIR__ . DIRECTORY_SEPARATOR . ''tab'';'
$new = '$baseDir = ''C:/TunerTools/Data/GuitarPro'';'

$count = ([regex]::Matches(
    $content,
    [regex]::Escape($old)
)).Count

if ($count -ne 2) {
    throw "Can dung 2 vi tri, nhung tim thay: $count"
}

$backup = "C:\TunerTools\Backups\api_songs_before_gp_" +
          (Get-Date -Format "yyyyMMdd_HHmmss") + ".php.bak"

Copy-Item -LiteralPath $file -Destination $backup -ErrorAction Stop

try {
    $content = $content.Replace($old, $new)

    [System.IO.File]::WriteAllBytes(
        $file,
        $encoding.GetBytes($content)
    )

    & "C:\php\php.exe" -l $file

    if ($LASTEXITCODE -ne 0) {
        throw "PHP syntax error."
    }

    Copy-Item -LiteralPath $file `
        -Destination $dest `
        -Force `
        -ErrorAction Stop

    Write-Host ""
    Write-Host "API_SONGS PATH UPDATED OK" -ForegroundColor Green
    Write-Host "Da sua: 2 vi tri"
    Write-Host "Guitar Pro: $gpFolder"
    Write-Host "Backup: $backup"
}
catch {
    Copy-Item -LiteralPath $backup `
        -Destination $file `
        -Force

    throw "Cap nhat that bai. Da khoi phuc PHP goc."
}
