    throw "Khong tim thay PHP chinh"
}

    Select-Object FullName, Length, LastWriteTime

$patterns = @(
    "auth_login",
    "permission_me",
    "action=list",
    "scandir",
    "is_dir",
    "DirectoryIterator",
    "Documents",
    "document_root",
    "DOCUMENT_ROOT"
)

    Select-Object LineNumber,
        @{Name="Keyword";Expression={$_.Matches[0].Value}} |
    Format-Table -AutoSize
