
# Thu muc tai file
$downloads = "$env:USERPROFILE\Downloads"
$extract = "$downloads\TunerTools_V5_FIX"
$install = "C:\TunerTools\Tools\PDF_TAB_GP_WEB"

# Tim file ZIP V5
$zip = Get-ChildItem $downloads -File |
    Where-Object { $_.Name -like "TunerTools_V5_FIX_20261113*.zip" } |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $zip) {
    Write-Host "LOI: Khong tim thay file ZIP V5 trong Downloads!" -ForegroundColor Red
    return
}

Write-Host "Tim thay: $($zip.FullName)" -ForegroundColor Green

# Giai nen
Expand-Archive -LiteralPath $zip.FullName `
    -DestinationPath $extract -Force -ErrorAction Stop

# Tim bo cai dat
$setup = Get-ChildItem $extract `
    -Filter "CAI_DAT_V5_FIX.ps1" -Recurse |
    Select-Object -First 1

if (-not $setup) {
    throw "Khong tim thay CAI_DAT_V5_FIX.ps1"
}

# Cai dat V5
powershell.exe -NoProfile -ExecutionPolicy Bypass `
    -File $setup.FullName

if ($LASTEXITCODE -ne 0) {
    throw "Cai dat V5 that bai. Kiem tra thong bao phia tren."
}

# Khoi dong website
& "$install\CHAY_WEB.bat"
