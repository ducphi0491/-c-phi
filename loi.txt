
$downloads = "$env:USERPROFILE\Downloads"
$dir = "C:\TunerTools\Tools\PDF_TAB_GP_WEB"

# Tu dong tim file ZIP V5
$zip = Get-ChildItem -Path $downloads `
    -Filter "TunerTools_PDF_GP_WEB_V5_DOI_CHIEU*.zip" `
    -File |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

if (-not $zip) {
    Write-Host "LOI: Khong tim thay file V5 trong Downloads." -ForegroundColor Red
    Write-Host "Hay tai file ZIP V5 ve truoc khi chay."
    return
}

Write-Host "Tim thay: $($zip.FullName)" -ForegroundColor Green

# Giai nen V5
New-Item -ItemType Directory -Path $dir -Force | Out-Null

Expand-Archive -LiteralPath $zip.FullName `
    -DestinationPath $dir -Force -ErrorAction Stop

Write-Host "Da giai nen V5 thanh cong." -ForegroundColor Green

# Kiem tra file khoi dong
$launcher = Join-Path $dir "CHAY_WEB.bat"

if (-not (Test-Path $launcher)) {
    throw "Khong tim thay CHAY_WEB.bat"
}

# Khoi dong website
& $launcher
