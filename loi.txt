$urls = @(
    "https://api.tunertools.top/api/get_lyrics",
    "http://127.0.0.1:5000/api/get_lyrics"
)

foreach ($url in $urls) {
    Write-Host "`nTEST: $url" -ForegroundColor Cyan

    & curl.exe -sS --max-time 15 `
        -X POST `
        -d "" `
        -o NUL `
        -w "HTTP STATUS: %{http_code}`n" `
        $url
}
