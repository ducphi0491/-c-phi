$ErrorActionPreference = "Stop"

$main = "C:\TunerTools\Data\GuitarPro"
$partial = "C:\TunerTools\Data\GuitarPro_COPY_INCOMPLETE"

# Xac nhan du lieu chinh van ton tai
if (-not (Test-Path -LiteralPath $main -PathType Container)) {
    throw "KHONG TIM THAY GUITAR PRO CHINH. DUNG XOA."
}

# Kiem tra dung thu muc ban sao
$item = Get-Item -LiteralPath $partial -Force -ErrorAction Stop

if (-not $item.PSIsContainer -or
    $item.FullName -ne $partial -or
    ($item.Attributes -band [IO.FileAttributes]::ReparsePoint)) {
    throw "THU MUC KHONG HOP LE. DUNG XOA."
}

Write-Host "DANG XOA BAN COPY DO 10GB..." -ForegroundColor Yellow

# Chi xoa ban copy do
Remove-Item -LiteralPath $partial -Recurse -Force -ErrorAction Stop

Write-Host "`nDA XOA THANH CONG!" -ForegroundColor Green

Write-Host "GUITAR PRO CHINH: $(Test-Path -LiteralPath $main)"
Write-Host "BAN COPY DO CON: $(Test-Path -LiteralPath $partial)"

Get-PSDrive C |
Select-Object Name,
@{Name="FreeGB";Expression={[math]::Round($_.Free/1GB,2)}} |
Format-Table -AutoSize
