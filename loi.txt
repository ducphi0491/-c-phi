$names = @(
    "api_songs.php",
    "api_gp_search.php",
    "api_documents.php"
)

foreach ($name in $names) {
    $old = "C:\inetpub\wwwroot\$name"
    $new = "C:\TunerTools\Web\API\$name"

    $isWrapper = $false

    if (Test-Path -LiteralPath $old) {
        $content = [IO.File]::ReadAllText($old)

        $expected = "C:/TunerTools/Web/API/$name"

        $isWrapper = $content.Contains($expected)
    }

    [PSCustomObject]@{
        API = $name
        PHP_Old_Exists = Test-Path -LiteralPath $old
        Wrapper_OK = $isWrapper
        PHP_New_Exists = Test-Path -LiteralPath $new
    }
} | Format-Table -AutoSize
