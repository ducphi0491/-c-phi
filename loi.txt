-- TunerTools - repair missing Sync V2 triggers
-- Source: existing ALTER_song_sync_v2.sql trigger definitions.
-- This script ONLY creates the 3 missing triggers on `tuner`.`songs`.
-- It does NOT change songs data, delete databases, add indexes, or alter PHP/token.
-- Run ONCE in MySQL Workbench only after confirming SHOW TRIGGERS has no rows.
-- Make a DB backup before running schema changes on production.

USE tuner;

DELIMITER $$

CREATE TRIGGER trg_songs_sync_after_insert
AFTER INSERT ON songs
FOR EACH ROW
BEGIN
    IF COALESCE(NEW.status, 0) = 1 THEN
        INSERT INTO song_sync_changes (song_id, operation)
        VALUES (NEW.id, 'upsert');
    END IF;
END$$

CREATE TRIGGER trg_songs_sync_after_update
AFTER UPDATE ON songs
FOR EACH ROW
BEGIN
    IF NOT (OLD.id <=> NEW.id) THEN
        IF COALESCE(OLD.status, 0) = 1 THEN
            INSERT INTO song_sync_changes (song_id, operation)
            VALUES (OLD.id, 'delete');
        END IF;
        IF COALESCE(NEW.status, 0) = 1 THEN
            INSERT INTO song_sync_changes (song_id, operation)
            VALUES (NEW.id, 'upsert');
        END IF;
    ELSEIF COALESCE(OLD.status, 0) = 1 AND COALESCE(NEW.status, 0) <> 1 THEN
        INSERT INTO song_sync_changes (song_id, operation)
        VALUES (OLD.id, 'delete');
    ELSEIF COALESCE(NEW.status, 0) = 1 AND (
        COALESCE(OLD.status, 0) <> COALESCE(NEW.status, 0)
        OR NOT (OLD.title <=> NEW.title)
        OR NOT (OLD.singerNames <=> NEW.singerNames)
        OR NOT (OLD.authorName <=> NEW.authorName)
        OR NOT (OLD.originalKey <=> NEW.originalKey)
        OR NOT (OLD.rhythm <=> NEW.rhythm)
        OR NOT (OLD.bpm <=> NEW.bpm)
        OR NOT (OLD.youtubeUrl <=> NEW.youtubeUrl)
        OR NOT (OLD.lyrics <=> NEW.lyrics)
        OR NOT (OLD.chordList <=> NEW.chordList)
        OR NOT (OLD.slug <=> NEW.slug)
        OR NOT (OLD.image_count <=> NEW.image_count)
        OR NOT (OLD.id_zing_mp3 <=> NEW.id_zing_mp3)
    ) THEN
        INSERT INTO song_sync_changes (song_id, operation)
        VALUES (NEW.id, 'upsert');
    END IF;
END$$

CREATE TRIGGER trg_songs_sync_after_delete
AFTER DELETE ON songs
FOR EACH ROW
BEGIN
    IF COALESCE(OLD.status, 0) = 1 THEN
        INSERT INTO song_sync_changes (song_id, operation)
        VALUES (OLD.id, 'delete');
    END IF;
END$$

DELIMITER ;

-- Should show exactly 3 trigger rows (INSERT, UPDATE, DELETE).
SHOW TRIGGERS WHERE `Table` = 'songs';

-- IMPORTANT: the change log remains empty until a NEW change occurs.
-- Creating triggers does not automatically backfill past missed updates.
SELECT COUNT(*) AS change_log_rows,
       COALESCE(MAX(change_id), 0) AS current_cursor
FROM song_sync_changes;
