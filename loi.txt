$root = "C:\inetpub\wwwroot"
$api = "C:\TunerTools\Web\API"

Write-Host "`n=== KIEM TRA TAI NGUYEN ===" -ForegroundColor Cyan

$paths = @(
    "$root\firebase-service-account.json",
    "$root\google-services.json",
    "$root\support_chat_uploads"
)

foreach ($path in $paths) {
    $exists = Test-Path -LiteralPath $path

    [PSCustomObject]@{
        Name   = Split-Path $path -Leaf
        Exists = $exists
        Type   = if ($exists) {
            if (Test-Path -LiteralPath $path -PathType Container) {
                "Folder"
            } else {
                "File"
            }
        } else {
            "Missing"
        }
    }
}

Write-Host "`n=== KIEM TRA THU MUC UPLOAD ===" -ForegroundColor Cyan

$upload = "$root\support_chat_uploads"

if (Test-Path -LiteralPath $upload -PathType Container) {
    $stats = Get-ChildItem -LiteralPath $upload -Recurse -File |
        Measure-Object -Property Length -Sum

    Write-Host "SO FILE: $($stats.Count)"
    Write-Host "DUNG LUONG MB: $([math]::Round($stats.Sum / 1MB, 2))"
}

Write-Host "`n=== KIEM TRA THAM CHIEU TRONG PHP ===" -ForegroundColor Cyan

$files = @(
    "$api\api_gp_search.php",
    "$api\api_songs.php",
    "$root\api_documents.php"
)

$patterns = @(
    'firebase-service-account.json',
    'google-services.json',
    'support_chat_uploads',
    'tuner_images_'
)

Select-String -Path $files -SimpleMatch -Pattern $patterns |
    Select-Object @{
        Name = 'File'
        Expression = { Split-Path $_.Path -Leaf }
    }, LineNumber, Pattern |
    Format-Table -AutoSize

Write-Host "`nKIEM TRA HOAN TAT" -ForegroundColor Green
