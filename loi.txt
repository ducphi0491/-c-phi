$paths = @(
    "C:\",
    "C:\inetpub\wwwroot",
    "C:\AI_Music"
)

foreach ($path in $paths) {
    if (Test-Path $path) {
        Write-Host "`n========================================" -ForegroundColor Cyan
        Write-Host "THU MUC: $path" -ForegroundColor Yellow
        Write-Host "========================================"

        $folders = @(Get-ChildItem -LiteralPath $path -Directory -Force -ErrorAction SilentlyContinue)

        $folders |
            Sort-Object Name |
            Select-Object Name, FullName |
            Format-Table -AutoSize

        Write-Host "TONG SO THU MUC CAP 1: $($folders.Count)" -ForegroundColor Green
    }
}
