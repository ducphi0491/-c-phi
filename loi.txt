>>
>> $d = Invoke-RestMethod -Uri $url -TimeoutSec 120
>> $songs = @($d.songs)
>>
>> $audio = @($songs | Where-Object {
>>     $_.has_audio_cache -eq $true
>> })
>>

>>     $_.has_lyrics_cache -eq $true
>> })
>>
>> $audioOnly = @($songs | Where-Object {
>>     $_.has_audio_cache -eq $true -and
>>     $_.has_lyrics_cache -ne $true
