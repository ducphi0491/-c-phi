$oldPath = 'C:\AI_Music'
$newPath = 'C:\TunerTools\Audio\AI_Music'

$files = @(
    'C:\inetpub\wwwroot\api_AI.bat',
    'C:\inetpub\wwwroot\api_gp_search.php',
    'C:\inetpub\wwwroot\api_songs.php'
)

$encoding = [System.Text.Encoding]::GetEncoding(28591)
$timestamp = Get-Date -Format 'yyyyMMdd_HHmmss'

foreach ($file in $files) {

    $bytes = [System.IO.File]::ReadAllBytes($file)
    $content = $encoding.GetString($bytes)

    $count = ([regex]::Matches(
        $content,
        [regex]::Escape($oldPath)
    )).Count

    if ($count -gt 0) {

        $backup = "C:\TunerTools\Backups\" +
                  (Split-Path $file -Leaf) +
                  "." + $timestamp + ".bak"

        Copy-Item -LiteralPath $file -Destination $backup -ErrorAction Stop

        $content = $content.Replace($oldPath, $newPath)

        [System.IO.File]::WriteAllBytes(
            $file,
            $encoding.GetBytes($content)
        )

        Write-Host "UPDATED: $file ($count vi tri)" -ForegroundColor Green
    }
}
