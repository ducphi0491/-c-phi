SELECT
    id,
    title,
    singerNames,
    slug,
    image_count,
    updatedAt,
    status
FROM songs
WHERE title LIKE '%Người Tình Mùa Đông%';



SHOW TRIGGERS
WHERE `Table` = 'songs';


SELECT
    change_id,
    song_id,
    operation,
    changed_at
FROM song_sync_changes
ORDER BY change_id DESC
LIMIT 20;
