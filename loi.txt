$roots = @(
    "C:\inetpub\wwwroot",
    "C:\TunerTools\Web\API",
    "C:\TunerTools\Web\tunertools"
)

$extensions = @(
    ".php",
    ".config",
    ".bat",
    ".cmd",
    ".ps1"
)

$pattern = '(?i)inetpub[\\/]+wwwroot|document_cache|support_chat_uploads|img_sys|[\\/]libs[\\/]|[\\/]tunertools[\\/]'

Write-Host "`n=== QUET THAM CHIEU DUONG DAN CU ===" -ForegroundColor Cyan

$files = @(
    foreach ($root in $roots) {

        if (Test-Path -LiteralPath $root) {

            Get-ChildItem -LiteralPath $root `
                -Recurse -File -ErrorAction SilentlyContinue |
            Where-Object {
                $extensions -contains $_.Extension.ToLowerInvariant()
            }
        }
    }
)

$files = @($files | Sort-Object FullName -Unique)

$hits = @(
    $files | Select-String -Pattern $pattern |
    ForEach-Object {

        [PSCustomObject]@{
            File  = $_.Path
            Line  = $_.LineNumber
            Match = $_.Matches[0].Value
        }
    }
)

Write-Host "SOURCE FILES SCANNED: $($files.Count)"
Write-Host "REFERENCES FOUND: $($hits.Count)"

$hits |
    Select-Object -First 60 |
    Format-Table -AutoSize
